using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;

[Route("api/[controller]")]
[ApiController]
public class StudentController : ControllerBase
{
    private readonly AppDbContext _context;

    public StudentController(AppDbContext context)
    {
        _context = context;
    }

    // GET
    [HttpGet]
    public async Task<IActionResult> Get()
    {
        return Ok(await _context.Students.ToListAsync());
    }

    // GET BY ID
    [HttpGet("{id}")]
    public async Task<IActionResult> Get(int id)
    {
        var s = await _context.Students.FindAsync(id);
        return Ok(s);
    }

    // POST (FIXED 🔥)
    [HttpPost]
    public async Task<IActionResult> Post([FromBody] Student s)
    {
        if (s == null)
            return BadRequest("Invalid data");

        _context.Students.Add(s);
        await _context.SaveChangesAsync();

        return Ok(s);
    }

    // PUT (FIXED 🔥)
    [HttpPut("{id}")]
    public async Task<IActionResult> Put(int id, [FromBody] Student s)
    {
        var existing = await _context.Students.FindAsync(id);
        if (existing == null) return NotFound();

        existing.Name = s.Name;
        existing.Email = s.Email;
        existing.Mobile = s.Mobile;
        existing.Age = s.Age;
        existing.City = s.City;
        existing.Gender = s.Gender;

        await _context.SaveChangesAsync();
        return Ok();
    }

    // DELETE
    [HttpDelete("{id}")]
    public async Task<IActionResult> Delete(int id)
    {
        var s = await _context.Students.FindAsync(id);
        if (s == null) return NotFound();

        _context.Students.Remove(s);
        await _context.SaveChangesAsync();
        return Ok();
    }

    // SEARCH
    [HttpGet("search")]
    public async Task<IActionResult> Search(string name, string city)
    {
        var query = _context.Students.AsQueryable();

        if (!string.IsNullOrEmpty(name))
            query = query.Where(x => x.Name.Contains(name));

        if (!string.IsNullOrEmpty(city))
            query = query.Where(x => x.City == city);

        return Ok(await query.ToListAsync());
    }

    // EMAIL CHECK
    [HttpGet("checkEmail")]
    public IActionResult CheckEmail(string email)
    {
        bool exists = _context.Students.Any(x => x.Email == email);
        return Ok(new { exists });
    }
}

const api = "https://localhost:7121/api/student";

// VALIDATION (FIXED)
function validate(){
    let name = $("#name").val();
    let email = $("#email").val();
    let mobile = $("#mobile").val();
    let age = $("#age").val();
    let city = $("#city").val();
    let gender = $("input[name=gender]:checked").val();

        let emailPattern = new RegExp("^[^ ]+@@[^ ]+\\.[a-z]{2,3}$");

    if(name.length < 3){ alert("Name min 3"); return false; }
    if(!email.match(emailPattern)){ alert("Invalid email"); return false; }
    if(mobile.length != 10){ alert("Mobile 10 digits"); return false; }
    if(age < 18){ alert("Age 18+"); return false; }
    if(city == ""){ alert("Select city"); return false; }
    if(!gender){ alert("Select gender"); return false; }

    return true;
}

// SAVE (FIXED)
function save(){

    if(!validate()) return;

    let student = {
        id: 0,
        name: $("#name").val(),
        email: $("#email").val(),
        mobile: $("#mobile").val(),
        age: parseInt($("#age").val()),
        city: $("#city").val(),
        gender: $("input[name=gender]:checked").val()
    };

    $.ajax({
        url: api,
        type: "POST",
        contentType: "application/json",
        data: JSON.stringify(student),

        success: function(){
            alert("Saved Successfully!");
            window.location.href = "/Home/Index";
        },

        error: function(err){
            console.log(err.responseText);
            alert("Error saving!");
        }
    });
}
</script>

<script>
    const api = "https://localhost:7121/api/student";

    // 🔥 AUTO LOAD
    $(document).ready(function(){
        loadStudents();
    });

    function loadStudents(){
        let name = $("#search").val();
        let city = $("#filterCity").val();

        let url = (name || city)
            ? `${api}/search?name=${name}&city=${city}`
            : api;

        $.get(url, function(data){
            let rows = "";

            if(data.length === 0){
                rows = "<tr><td colspan='4'>No Data Found</td></tr>";
            } else {
                data.forEach(s=>{
                    rows += `<tr>
                        <td>${s.name}</td>
                        <td>${s.email}</td>
                        <td>${s.city}</td>
                        <td>
                            <button onclick="edit(${s.id})">Edit</button>
                            <button onclick="del(${s.id})">Delete</button>
                        </td>
                    </tr>`;
                });
            }

            $("#tableBody").html(rows);
        });
    }

    function del(id){
        if(confirm("Delete?")){
            $.ajax({
                url: api + "/" + id,
                type: "DELETE",
                success: loadStudents
            });
        }
    }

    function edit(id){
        window.location.href = "/Home/Create?id=" + id;
    }
</script>
